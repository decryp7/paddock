//! Round trip of one callback over paddock-ipc on a private endpoint pair, in one process.
use paddock_ipc::{endpoint, CallCtx, Connection, FrameLimit, Handler, Peer, PeerOptions, RemoteError};
use std::sync::Arc;
struct Files;
impl Handler for Files {
    fn call(&self, _: &CallCtx, body: &[u8]) -> Result<Vec<u8>, RemoteError> {
        let n = u32::from_le_bytes(body[..4].try_into().unwrap()) as usize;
        Ok(vec![7u8; n])
    }
}
fn main() {
    let (x, y) = endpoint::pair().unwrap();
    let limit = FrameLimit::default();
    let a = Peer::start(Connection::raw(x, limit), Arc::new(Files) as Arc<dyn Handler>, PeerOptions::new("a")).unwrap();
    let _b = Peer::start(Connection::raw(y, limit), Arc::new(Files) as Arc<dyn Handler>, PeerOptions::new("b")).unwrap();
    for (size, n) in [(512u32, 200_000u32), (4096, 100_000), (65536, 40_000), (1 << 20, 4000), (4 << 20, 1000)] {
        let t = std::time::Instant::now();
        for _ in 0..n { let r = a.call(size.to_le_bytes().to_vec(), None).unwrap(); assert_eq!(r.len(), size as usize); }
        let s = t.elapsed().as_secs_f64();
        println!("RESULT reply_bytes={size} calls={n} mean_us={:.1} mib_per_s={:.0}", s * 1e6 / n as f64, (size as f64 * n as f64 / (1 << 20) as f64) / s);
    }
}
